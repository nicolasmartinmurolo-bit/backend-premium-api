import os

OUTPUT_FILE = "proyecto_completo.txt"
VALID_EXTENSIONS = (".py", ".env", ".ini", ".md", ".json", ".txt")
EXCLUDE_DIRS = {".venv", "__pycache__", ".git", ".pytest_cache", "alembic/versions"}

def export_project():
    print("Iniciando empaquetado del proyecto...")
    count = 0
    with open(OUTPUT_FILE, "w", encoding="utf-8", errors="ignore") as outfile:
        for root, dirs, files in os.walk("."):
            # Excluimos carpetas no deseadas modificando dirs in-place
            dirs[:] = [d for d in dirs if d not in EXCLUDE_DIRS]
            
            for file in files:
                if file.endswith(VALID_EXTENSIONS) and file != OUTPUT_FILE:
                    file_path = os.path.join(root, file)
                    print(f"Procesando: {file_path}")
                    
                    outfile.write(f"\n\n{'='*80}\n")
                    outfile.write(f"ARCHIVO: {file_path}\n")
                    outfile.write(f"{'='*80}\n\n")
                    
                    try:
                        with open(file_path, "r", encoding="utf-8", errors="ignore") as infile:
                            outfile.write(infile.read())
                        count += 1
                    except Exception as e:
                        outfile.write(f"[Error al leer el archivo: {e}]\n")
                        
    print(f"\n¡Éxito total! Se empaquetaron {count} archivos en: {OUTPUT_FILE}")

if __name__ == "__main__":
    export_project()

def run_migrations_offline() -> None:
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url,
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online() -> None:
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """
    connectable = engine_from_config(
        config.get_section(config.config_ini_section, {}),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection, target_metadata=target_metadata
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

Sus responsabilidades principales son:

2. Configurar la ruta base de la documentación interactiva (Swagger UI).

"""
Módulo de Inyección de Dependencias y Seguridad (Autenticación y Roles).
--------------------------------------------------------------------------------
Este archivo agrupa las funciones de dependencia fundamentales utilizadas 
para proteger los endpoints de la API:
1. Configuración del esquema OAuth2 para la extracción de tokens Bearer.
2. Validador de identidad mediante tokens JWT (get_current_user).
3. Validador de privilegios de administrador / superusuario (get_current_active_superuser).
"""

from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from jose import jwt, JWTError
from sqlalchemy.orm import Session

from app.core.config import settings
from app.core.database import get_db  # <--- Importamos la única fuente oficial de get_db
from app.models.user import User as UserModel
from app.schemas.token import TokenData

# -------------------------------------------------------------------------
# 1. Esquema de Seguridad OAuth2 (Token Bearer)
# -------------------------------------------------------------------------
# OAuth2PasswordBearer le indica a FastAPI dónde debe buscar el token en las 
# peticiones entrantes. Apunta directamente a la ruta de login (/api/v1/login/access-token)
# para que Swagger UI sepa cómo y dónde autenticarse automáticamente.
reusable_oauth2 = OAuth2PasswordBearer(
    tokenUrl=f"{settings.API_V1_STR}/login/access-token"
)


# -------------------------------------------------------------------------
# 2. Dependencia de Usuario Actual (Autenticación por JWT)
# -------------------------------------------------------------------------
def get_current_user(
    db: Session = Depends(get_db), token: str = Depends(reusable_oauth2)
) -> UserModel:
    """
    Intercepta la petición HTTP, extrae y valida el token JWT recibido en los headers.
    Retorna el objeto del usuario autenticado si el token es válido y el usuario existe.
    Lanza una excepción HTTP 401 (Unauthorized) si algo falla.
    """
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="Could not validate credentials",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        # Decodificamos el token usando la clave secreta y el algoritmo global
        payload = jwt.decode(
            token, settings.SECRET_KEY, algorithms=[settings.ALGORITHM]
        )
        email: str = payload.get("sub")  # 'sub' (subject) almacena el email del usuario
        if email is None:
            raise credentials_exception
        token_data = TokenData(email=email)
    except JWTError:
        # Si el token expiró, fue alterado o es inválido, explota aquí
        raise credentials_exception
        
    # Buscamos al usuario en la base de datos PostgreSQL mediante su email
    user = db.query(UserModel).filter(UserModel.email == token_data.email).first()
    if user is None:
        raise credentials_exception
        
    # Validamos adicionalmente que la cuenta del usuario se encuentre activa
    if not user.is_active:
        raise HTTPException(status_code=400, detail="Inactive user")
        
    return user


# -------------------------------------------------------------------------
# 3. Dependencia de Control de Acceso por Roles (Superusuario)
# -------------------------------------------------------------------------
def get_current_active_superuser(
    current_user: UserModel = Depends(get_current_user),
) -> UserModel:
    """
    Capa de autorización superior. Toma al usuario ya autenticado (gracias a get_current_user) 
    y evalúa si cuenta con privilegios de administrador (is_superuser = True).
    Si no es administrador, bloquea el acceso lanzando un error HTTP 403 (Forbidden).
    """
    if not current_user.is_superuser:
        raise HTTPException(
            status_code=status.HTTP_403_FORBIDDEN, 
            detail="El usuario no cuenta con privilegios de administrador."
        )
    return current_user

"""
Endpoints de Autenticación y Login (OAuth2).
--------------------------------------------------------------------------------
Este módulo expone las rutas públicas relacionadas con la acreditación de identidad. 
Su función principal es recibir las credenciales del usuario (correo y contraseña), 
verificar su autenticidad contra la base de datos y emitir un token JWT cifrado 
que servirá como pase de acceso para las rutas protegidas de la API.
"""

from datetime import timedelta
from fastapi import APIRouter, Depends, HTTPException, status
from fastapi.security import OAuth2PasswordRequestForm
from sqlalchemy.orm import Session

from app.core.database import SessionLocal
from app.core.config import settings
from app.core.security import verify_password, create_access_token
from app.models.user import User as UserModel
from app.schemas.token import Token

# -------------------------------------------------------------------------
# 1. Definición del Enrutador (Router)
# -------------------------------------------------------------------------
# Instanciamos APIRouter para agrupar todas las rutas de autenticación 
# y permitir que el enrutador principal (api_router) las registre de forma modular.
router = APIRouter()

def get_db():
    """
    Dependencia local para abrir y cerrar sesiones de base de datos 
    durante el proceso de autenticación.
    """
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()


# -------------------------------------------------------------------------
# 2. Endpoint de Autenticación y Emisión de Tokens (Login)
# -------------------------------------------------------------------------
@router.post("/login/access-token", response_model=Token)
def login_access_token(
    db: Session = Depends(get_db), 
    form_data: OAuth2PasswordRequestForm = Depends()
):
    """
    Endpoint compatible con OAuth2 para el inicio de sesión.
    - Recibe un formulario estándar (OAuth2PasswordRequestForm) donde el campo 
      'username' corresponde al correo electrónico del usuario y 'password' a su clave.
    - Retorna un token de acceso cifrado (JWT) junto con su tipo (bearer).
    """
    
    # Paso 1: Buscar al usuario en PostgreSQL utilizando su correo electrónico.
    # Nota: OAuth2PasswordRequestForm almacena el identificador en la propiedad 'username'.
    user = db.query(UserModel).filter(UserModel.email == form_data.username).first()
    
    # Paso 2: Verificar la existencia del usuario y la validez de la contraseña.
    # Comparamos la contraseña en texto plano con el hash seguro guardado usando bcrypt.
    if not user or not verify_password(form_data.password, user.hashed_password):
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Incorrect email or password",
        )
    
    # Paso 3: Validar que el estado de la cuenta del usuario se encuentre activo.
    if not user.is_active:
        raise HTTPException(
            status_code=status.HTTP_400_BAD_REQUEST,
            detail="Inactive user",
        )
    
    # Paso 4: Configurar el tiempo de vida útil del token de acceso 
    # utilizando los parámetros globales definidos en el archivo .env y config.py.
    access_token_expires = timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    
    # Paso 5: Generar y retornar el token JWT incluyendo el email como identificador (subject 'sub').
    return {
        "access_token": create_access_token(
            data={"sub": user.email}, expires_delta=access_token_expires
        ),

"""
Endpoints de Gestión de Usuarios (CRUD y Perfil).
--------------------------------------------------------------------------------
Este módulo concentra las rutas HTTP destinadas a la administración de usuarios:
1. Registro de nuevos usuarios (POST /) con cifrado automático de contraseñas.
2. Listado general de usuarios (GET /) protegido exclusivamente para administradores.
3. Consulta de perfil propio (GET /me) validando la identidad mediante el token JWT.
"""

from typing import List
from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy.orm import Session

from app.core.database import get_db  # Dependencia centralizada de sesión de base de datos
from app.core.security import get_password_hash
from app.models.user import User as UserModel
from app.schemas.user import User, UserCreate
from app.api.deps import get_current_user, get_current_active_superuser

# -------------------------------------------------------------------------
# 1. Definición del Enrutador de Usuarios
# -------------------------------------------------------------------------
router = APIRouter()


# -------------------------------------------------------------------------
# 2. Endpoint de Registro de Usuarios (Create / POST)
# -------------------------------------------------------------------------
@router.post("/", response_model=User)
def create_user(user_in: UserCreate, db: Session = Depends(get_db)):
    """
    Registra un nuevo usuario en la base de datos:
    - Valida que el correo electrónico no esté previamente registrado.
    - Toma la contraseña en texto plano y genera su hash seguro con bcrypt.
    - Almacena el registro en PostgreSQL de forma transaccional (commit/rollback).
    """
    
    # Paso 1: Verificamos si ya existe un usuario con el mismo email en la base de datos.
    existing_user = db.query(UserModel).filter(UserModel.email == user_in.email).first()
    if existing_user:
        raise HTTPException(
            status_code=400,
            detail="El correo electrónico ya está registrado en la base de datos."
        )
    
    # Paso 2: Creamos una instancia del modelo ORM User con los datos validados por Pydantic.
    # Nota clave: La contraseña NUNCA se guarda en texto plano; se usa get_password_hash().
    db_user = UserModel(
        email=user_in.email,
        hashed_password=get_password_hash(user_in.password),
        full_name=user_in.full_name,
        is_active=user_in.is_active,
        is_superuser=user_in.is_superuser
    )
    
    # Paso 3: Intentamos guardar los cambios en la base de datos de forma segura (Transacción).
    try:
        db.add(db_user)
        db.commit()
        db.refresh(db_user)  # Actualiza la instancia con el ID generado automáticamente por la BD.
        return db_user
    except Exception as e:
        # Si ocurre cualquier fallo inesperado, revertimos la transacción para no corromper la BD.
        db.rollback()
        print(f"ERROR REAL EN BD: {str(e)}")
        raise HTTPException(status_code=400, detail=f"Error: {str(e)}")

    - Requiere obligatoriamente que el usuario que hace la petición tenga el rol

"""

from sqlalchemy.orm import declarative_base, sessionmaker
from app.core.config import settings

# 'postgresql://' pero no especifica explícitamente el driver de psycopg2, 
# se realiza un reemplazo automático para evitar errores de compatibilidad con SQLAlchemy.
if "postgresql://" in db_url and "psycopg2" not in db_url:
    db_url = db_url.replace("postgresql://", "postgresql+psycopg2://")

# 2. Creación del Motor de Base de Datos (Engine)
# -------------------------------------------------------------------------
try:
    # create_engine es el punto central de conexión de SQLAlchemy.
    # 'connect_args' se utiliza para pasar parámetros específicos del driver a nivel 
    # de conexión en bruto, como un tiempo límite (timeout) de 5 segundos para evitar 
    # que el servidor se quede bloqueado indefinidamente si la base de datos no responde.
    engine = create_engine(
        db_url,
        connect_args={"connect_timeout": 5}
    )
    print("LOG: Motor de Base de Datos configurado correctamente.")
except Exception as e:
    # Capturamos cualquier error crítico al intentar inicializar el motor y lo relanzamos 
    # para detener el inicio de la aplicación si no hay acceso a los datos.
    print(f"LOG ERROR: Falló la creación del motor de BD: {e}")
    raise e

# -------------------------------------------------------------------------
# 3. Configuración de Sesiones y Modelos ORM
# -------------------------------------------------------------------------

# Fábrica de sesiones (SessionLocal):
# Cada instancia de SessionLocal representará una "sesión de trabajo" temporal 
# con la base de datos. Desactivamos autocommit y autoflush para mantener un 
# control manual y transaccional estricto de los cambios (commits y rollbacks).
SessionLocal = sessionmaker(autocommit=False, autoflush=False, bind=engine)

# Base declarativa:
# Clase constructora de la cual heredarán todos nuestros modelos ORM (como User).
# Permite mapear clases de Python directamente a tablas relacionales en la base de datos.
Base = declarative_base()

# -------------------------------------------------------------------------
# 4. Inyección de Dependencias (Ciclo de Vida de la Sesión)
# -------------------------------------------------------------------------
def get_db():
    """
    Dependencia global generadora (Generator).
    Se encarga de abrir una nueva sesión de base de datos para cada petición HTTP

    Esto previene fugas de memoria y conexiones colgadas en el servidor.
    """
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()

"""
Módulo de Seguridad, Cifrado y Autenticación por Tokens (Bcrypt y JWT).
--------------------------------------------------------------------------------
Este archivo concentra toda la lógica criptográfica de la aplicación:
1. El hashing unidireccional de contraseñas utilizando la librería 'bcrypt' 
   de forma directa (evitando conflictos de compatibilidad).
2. La generación, firma y validación de tokens digitales (JSON Web Tokens - JWT) 
   utilizados para implementar una autenticación sin estado (stateless).
"""

import bcrypt
from datetime import datetime, timedelta
from typing import Optional
from jose import jwt
from app.core.config import settings

# -------------------------------------------------------------------------
# 1. Funciones de Hashing y Verificación de Contraseñas (Bcrypt)
# -------------------------------------------------------------------------

def verify_password(plain_password: str, hashed_password: str) -> bool:
    """
    Verifica si una contraseña en texto plano ingresada por el usuario 
    coincide con el hash seguro almacenado en la base de datos.
    Retorna True si son idénticas o False en caso contrario.
    """
    return bcrypt.checkpw(
        plain_password.encode('utf-8'), 
        hashed_password.encode('utf-8')
    )

def get_password_hash(password: str) -> str:
    """
    Genera un hash criptográfico irreversible de la contraseña utilizando bcrypt.
    
    Nota de seguridad: Bcrypt tiene una limitación nativa de 72 bytes máximos.
    Por ello, convertimos el texto a bytes y aplicamos un corte preventivo a los 
    primeros 72 bytes si la contraseña es excesivamente larga, evitando errores de sistema.
    """
    pwd_bytes = password.encode('utf-8')
    if len(pwd_bytes) > 72:
        pwd_bytes = pwd_bytes[:72]
        
    # Generamos la semilla aleatoria (salt) y aplicamos el hash con bcrypt
    hashed = bcrypt.hashpw(pwd_bytes, bcrypt.gensalt())
    return hashed.decode('utf-8')


# -------------------------------------------------------------------------
# 2. Generación de Tokens de Acceso (JSON Web Tokens - JWT)
# -------------------------------------------------------------------------

def create_access_token(data: dict, expires_delta: Optional[timedelta] = None) -> str:
    """
    Genera un token JWT firmado digitalmente para autenticar al usuario en peticiones futuras.
    
    Parámetros:
    - data: Diccionario con la carga útil o 'payload' (ej: {"sub": user.email}).
    - expires_delta: Tiempo de vida opcional para el token. Si no se provee, 
      utiliza por defecto el valor configurado en ACCESS_TOKEN_EXPIRE_MINUTES.
    """
    to_encode = data.copy()
    
    # Determinamos el tiempo exacto de expiración (ahora + delta de tiempo)
    if expires_delta:
        expire = datetime.utcnow() + expires_delta
    else:
        expire = datetime.utcnow() + timedelta(minutes=settings.ACCESS_TOKEN_EXPIRE_MINUTES)
    
    # Inyectamos la fecha de expiración estándar ('exp') dentro del payload del token
    to_encode.update({"exp": expire})
    
    # Codificamos el token usando la clave secreta global y el algoritmo de firma (ej. HS256)
    encoded_jwt = jwt.encode(to_encode, settings.SECRET_KEY, algorithm=settings.ALGORITHM)
    
    return encoded_jwt
